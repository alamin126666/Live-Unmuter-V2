━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

```python
from aiogram.types import InlineKeyboardMarkup, InlineKeyboardButton

# ব্লু বাটন (Primary - মেইন অ্যাকশন)
InlineKeyboardButton(
    text="লগিন করুন",
    callback_data="login",
    style="primary"  # ব্লু কালার
)

# গ্রিন বাটন (Success - অ্যাকশন)
InlineKeyboardButton(
    text="START BOT",
    callback_data="run",
    style="success"  # গ্রিন কালার
)

# রেড বাটন (Danger - ব怖ত ações)
InlineKeyboardButton(
    text="STOP BOT",
    callback_data="stop",
    style="danger"  # রেড কালার
)
```

তাই কি ButtonStyle এনাম ব্যবহার করা যায়:

```python
from aiogram.enums import ButtonStyle

InlineKeyboardButton(
    text="START BOT",
    callback_data="run",
    style=ButtonStyle.SUCCESS  # অথবা style="success"
)
```
